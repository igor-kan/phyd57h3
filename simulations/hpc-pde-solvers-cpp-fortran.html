1. Interactive 2D Heat Diffusion Simulation (Observable JS)

Observe 2D numerical PDE relaxation in real time on an HTML5 Canvas. The simulation solves the parabolic heat equation: \[\frac{\partial u}{\partial t} = \alpha \left( \frac{\partial^2 u}{\partial x^2} + \frac{\partial^2 u}{\partial y^2} \right)\] using a 5-point finite difference stencil directly inside your web browser:

Code 
viewof diffusivity = Inputs.range([0.05, 0.45], {value: 0.22, step: 0.01, label: "Thermal Diffusivity (α Δt / Δx²)"})
viewof sourceTemp = Inputs.range([50, 500], {value: 300, step: 10, label: "Central Source Temperature (°C)"})
Code 
heatCanvas = {
  const N = 60;
  const size = 360;
  const ctx = DOM.context2d(size, size);
  const cell = size / N;

  // Initialize grid
  let u = Array.from({length: N}, () => new Float32Array(N));
  let u_new = Array.from({length: N}, () => new Float32Array(N));

  function step() {
    const alpha = diffusivity;

    // Apply continuous heat source in the center
    const mid = Math.floor(N / 2);
    for (let i = mid - 3; i <= mid + 3; i++) {
      for (let j = mid - 3; j <= mid + 3; j++) {
        u[i][j] = sourceTemp;
      }
    }

    // 5-point stencil update
    for (let i = 1; i < N - 1; i++) {
      for (let j = 1; j < N - 1; j++) {
        u_new[i][j] = u[i][j] + alpha * (
          u[i+1][j] + u[i-1][j] + u[i][j+1] + u[i][j-1] - 4 * u[i][j]
        );
      }
    }

    // Copy back
    for (let i = 1; i < N - 1; i++) {
      for (let j = 1; j < N - 1; j++) {
        u[i][j] = u_new[i][j];
      }
    }

    // Render heatmap
    for (let i = 0; i < N; i++) {
      for (let j = 0; j < N; j++) {
        const val = Math.min(1.0, u[i][j] / 500);
        // Colormap (inferno-like)
        const r = Math.floor(255 * Math.sin(val * Math.PI * 0.5));
        const g = Math.floor(255 * Math.pow(val, 2));
        const b = Math.floor(255 * (1 - Math.cos(val * Math.PI * 0.5)) * 0.4);
        ctx.fillStyle = `rgb(${r},${g},${b})`;
        ctx.fillRect(i * cell, j * cell, cell, cell);
      }
    }
  }

  const timer = d3.interval(step, 30);
  invalidation.then(() => timer.stop());

  return ctx.canvas;
}

2. Multi-Threaded C++17 OpenMP Red-Black SOR Solver

Gauss-Seidel with Successive Over-Relaxation (SOR) achieves optimal convergence for elliptic PDEs like Poisson’s equation \(\nabla^2 \phi = -\rho / \epsilon_0\). Red-Black checkerboard coloring eliminates data race conditions, allowing perfect multi-core parallel scaling:

// poisson_sor_openmp.cpp
// Compile: g++ -O3 -fopenmp -march=native -std=c++17 poisson_sor_openmp.cpp -o sor_solver
#include <iostream>
#include <vector>
#include <cmath>
#include <omp.h>

const int N = 512;
const double TOLERANCE = 1e-6;
const int MAX_ITER = 10000;

int main() {
    std::vector<std::vector<double>> phi(N, std::vector<double>(N, 0.0));
    std::vector<std::vector<double>> rho(N, std::vector<double>(N, 0.0));

    // Place point charges
    rho[N/3][N/3] = 100.0;
    rho[2*N/3][2*N/3] = -100.0;

    double h = 1.0 / (N - 1);
    // Optimal over-relaxation parameter: omega = 2 / (1 + sin(pi * h))
    double omega = 2.0 / (1.0 + std::sin(M_PI * h));

    std::cout << "Running Red-Black SOR with " << omp_get_max_threads() 
              << " OpenMP threads. Omega = " << omega << std::endl;

    double diff = 1.0;
    int iter = 0;

    while (diff > TOLERANCE && iter < MAX_ITER) {
        diff = 0.0;

        // 1. Red points: (i + j) % 2 == 0
        #pragma omp parallel for reduction(max:diff) collapse(2)
        for (int i = 1; i < N - 1; ++i) {
            for (int j = 1; j < N - 1; ++j) {
                if ((i + j) % 2 == 0) {
                    double target = 0.25 * (phi[i+1][j] + phi[i-1][j] + phi[i][j+1] + phi[i][j-1] + h*h*rho[i][j]);
                    double delta = std::abs(target - phi[i][j]);
                    if (delta > diff) diff = delta;
                    phi[i][j] = (1.0 - omega) * phi[i][j] + omega * target;
                }
            }
        }

        // 2. Black points: (i + j) % 2 == 1
        #pragma omp parallel for reduction(max:diff) collapse(2)
        for (int i = 1; i < N - 1; ++i) {
            for (int j = 1; j < N - 1; ++j) {
                if ((i + j) % 2 == 1) {
                    double target = 0.25 * (phi[i+1][j] + phi[i-1][j] + phi[i][j+1] + phi[i][j-1] + h*h*rho[i][j]);
                    double delta = std::abs(target - phi[i][j]);
                    if (delta > diff) diff = delta;
                    phi[i][j] = (1.0 - omega) * phi[i][j] + omega * target;
                }
            }
        }

        iter++;
    }

    std::cout << "Converged in " << iter << " iterations with max residual = " << diff << std::endl;
    return 0;
}

3. Fortran 90: High-Performance 2D Advection Solver

! advection_2d.f90
! Compile: gfortran -O3 -std=f2018 advection_2d.f90 -o advection_sim
program advection_solver
    implicit none
    integer, parameter :: N = 256
    real(8) :: u(0:N, 0:N), u_new(0:N, 0:N)
    real(8) :: c_x, c_y, dx, dy, dt
    integer :: i, j, step, n_steps

    dx = 1.0d0 / dble(N)
    dy = 1.0d0 / dble(N)
    c_x = 1.0d0; c_y = 1.0d0
    dt = 0.4d0 * dx / sqrt(c_x**2 + c_y**2) ! CFL stability criterion

    ! Initialize 2D Gaussian pulse
    do j = 0, N
        do i = 0, N
            u(i, j) = exp(-((i*dx - 0.3d0)**2 + (j*dy - 0.3d0)**2) / 0.02d0)
        end do
    end do

    n_steps = 500
    ! Lax-Wendroff 2D advection scheme
    do step = 1, n_steps
        do j = 1, N - 1
            do i = 1, N - 1
                u_new(i, j) = u(i, j) - 0.5d0 * (c_x * dt / dx) * (u(i+1, j) - u(i-1, j)) &
                                      - 0.5d0 * (c_y * dt / dy) * (u(i, j+1) - u(i, j-1))
            end do
        end do
        u = u_new
    end do

    print *, "2D Advection successfully integrated for ", n_steps, " time steps."
end program advection_solver

## 2. Multi-Threaded C++17 OpenMP Red-Black SOR Solver

              << " OpenMP threads. Omega = " << omega << std::endl;
